Include(FetchContent)

# Disable clang-tidy for external projects.
set(MY_ORIGINAL_CLANG_TIDY_CMD ${CMAKE_CXX_CLANG_TIDY})
set(CMAKE_CXX_CLANG_TIDY "")

FetchContent_Declare(
    Catch2
    SYSTEM
    GIT_REPOSITORY https://github.com/catchorg/Catch2.git
    GIT_TAG v3.3.0
)

FetchContent_MakeAvailable(Catch2)

# Add Catch2 headers to system include directories. This suppresses
# clang-tidy warnings for these headers for CMake versions < 3.25
# which don't support the SYSTEM argument to add_subdirectory().
get_target_property(CATCH2_IID Catch2 INTERFACE_INCLUDE_DIRECTORIES)
set_target_properties(Catch2 PROPERTIES INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "${CATCH2_IID}")

target_compile_options(Catch2 PUBLIC -Wno-psabi)

# Re-enable clang-tidy for our projects.
set(CMAKE_CXX_CLANG_TIDY ${MY_ORIGINAL_CLANG_TIDY_CMD})


add_executable(tests
    distributions/bernoulli.cpp
    distributions/binomial.cpp
    distributions/cauchy.cpp
    distributions/exponential.cpp
    distributions/gaussian.cpp
    distributions/geometric.cpp
    distributions/gumbel.cpp
    distributions/laplace.cpp
    distributions/lognormal.cpp
    distributions/poisson.cpp
    distributions/rayleigh.cpp
    distributions/uniform.cpp
    distributions/weibull.cpp
    services/copy_stream_state.cpp
    services/copy_stream.cpp
    services/delete_stream.cpp
    services/get_brng_properties.cpp
    services/get_stream_state_brng.cpp
    services/leapfrog.cpp
    services/leapfrog_quasi.cpp
    services/new_stream_ex.cpp
    services/new_stream.cpp
    services/nondeterm.cpp
    services/skip_ahead_ex.cpp
    services/skip_ahead.cpp
    services/uniform_bit32.cpp
    services/uniform_bit64.cpp
    services/save_load_state.cpp
)
target_link_libraries(tests PRIVATE
  refng
  vsl
  Catch2::Catch2WithMain
)
target_include_directories(tests PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}
)

# Create the supports_autogen.h header at configure time.
set(CREATE_SUPPORT_DEFINES_DIR ${OPENRNG_ROOT}/tools/create_support_defines)
set(CREATE_SUPPORT_DEFINES_SCRIPT ${CREATE_SUPPORT_DEFINES_DIR}/create_support_defines.py)
set(INPUT_JSON ${CREATE_SUPPORT_DEFINES_DIR}/generators.json)
set(OUTPUT_HEADER ${CMAKE_CURRENT_BINARY_DIR}/supports_autogen.h)
execute_process(
    COMMAND python3 ${CREATE_SUPPORT_DEFINES_SCRIPT}
    INPUT_FILE ${INPUT_JSON}
    OUTPUT_FILE ${OUTPUT_HEADER}
    RESULT_VARIABLE CREATE_SUPPORT_DEFINES_STATUS
    ERROR_VARIABLE CREATE_SUPPORT_DEFINES_ERROR
)

if (NOT CREATE_SUPPORT_DEFINES_STATUS EQUAL 0)
  message(FATAL_ERROR "Failed to create support defines:\n${CREATE_SUPPORT_DEFINES_ERROR}")
endif()

# Rerun cmake configure if create_support_defines or input json change.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${INPUT_JSON} ${CREATE_SUPPORT_DEFINES_SCRIPT})

# Tell the compiler to search for autogenerated headers in the build directory.
target_include_directories(tests PRIVATE ${CMAKE_CURRENT_BINARY_DIR})

list(APPEND CMAKE_MODULE_PATH ${catch2_SOURCE_DIR}/extras)
include(Catch)
# Add --allow-running-no-tests to invocation so that skips are not reported as failures
catch_discover_tests(tests TEST_SPEC ${OPENRNG_TEST_SPEC} EXTRA_ARGS --allow-running-no-tests)

# check-copyright-headers relies on being in a git repository
if (GIT_HASH)
  add_test(NAME check-copyright-headers
      COMMAND python3 ${OPENRNG_ROOT}/tools/copyright_header/copyright_header.py
      WORKING_DIRECTORY ${OPENRNG_ROOT}
  )
endif()
